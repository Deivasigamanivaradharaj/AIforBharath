# data/scripts/etl_pipeline.pyimport pandas as pdimport boto3from sqlalchemy import create_engineimport logginglogging.basicConfig(level=logging.INFO)logger = logging.getLogger(__name__)class DataPipeline:    def __init__(self):        self.s3_client = boto3.client('s3')        self.db_engine = create_engine('postgresql://admin:password@endpoint:5432/sellersense')        def extract_from_s3(self, bucket, key):        """Extract data from S3"""        logger.info(f"Extracting data from s3://{bucket}/{key}")        obj = self.s3_client.get_object(Bucket=bucket, Key=key)        return pd.read_csv(obj['Body'])        def transform_sales_data(self, df):        """Clean and transform sales data"""        logger.info("Transforming sales data")        df['date'] = pd.to_datetime(df['date'])        df['sales'] = df['sales'].fillna(0)        df['revenue'] = df['sales'] * df['price']                # Add derived features        df['day_of_week'] = df['date'].dt.dayofweek        df['month'] = df['date'].dt.month        df['is_weekend'] = df['day_of_week'].isin([5, 6])                return df        def load_to_rds(self, df, table_name):        """Load transformed data to RDS"""        logger.info(f"Loading data to table: {table_name}")        df.to_sql(table_name, self.db_engine, if_exists='replace', index=False)        logger.info(f"Successfully loaded {len(df)} records")    def run_pipeline(self):        """Execute the complete ETL pipeline"""        try:            # Extract            raw_data = self.extract_from_s3('sellersense-raw-data-bucket', 'sales_data.csv')                        # Transform            processed_data = self.transform_sales_data(raw_data)                        # Load            self.load_to_rds(processed_data, 'sales_data')                        logger.info("ETL pipeline completed successfully")                    except Exception as e:            logger.error(f"ETL pipeline failed: {str(e)}")            raiseif __name__ == "__main__":    pipeline = DataPipeline()    pipeline.run_pipeline()

# models/forecasting/demand_forecaster.pyimport pandas as pdfrom prophet import Prophetimport joblibimport boto3from sklearn.metrics import mean_absolute_error, mean_squared_errorimport numpy as npclass DemandForecaster:    def __init__(self):        self.model = Prophet(            yearly_seasonality=True,             weekly_seasonality=True,            daily_seasonality=False,            changepoint_prior_scale=0.05        )        self.s3_client = boto3.client('s3')        self.model_bucket = 'sellersense-model-artifacts-bucket'        def prepare_data(self, df):        """Prepare data for Prophet model"""        # Prophet requires 'ds' and 'y' columns        prophet_df = df[['date', 'sales']].rename(columns={'date': 'ds', 'sales': 'y'})        prophet_df = prophet_df.sort_values('ds')        return prophet_df        def add_custom_seasonalities(self):        """Add custom seasonalities for Indian market"""        # Add Diwali effect (October-November)        self.model.add_seasonality(name='diwali', period=365.25, fourier_order=2)                # Add monsoon effect (June-September)        self.model.add_seasonality(name='monsoon', period=365.25, fourier_order=2)        def train_model(self, df):        """Train Prophet model"""        prophet_df = self.prepare_data(df)                # Add custom seasonalities        self.add_custom_seasonalities()                # Train model        self.model.fit(prophet_df)                # Save model to S3        model_path = '/tmp/prophet_model.pkl'        joblib.dump(self.model, model_path)                self.s3_client.upload_file(            model_path,             self.model_bucket,             'models/forecasting/prophet_model.pkl'        )                return self.model        def predict(self, periods=90):        """Generate predictions"""        future = self.model.make_future_dataframe(periods=periods)        forecast = self.model.predict(future)                return forecast[['ds', 'yhat', 'yhat_lower', 'yhat_upper']].tail(periods)        def evaluate_model(self, test_df):        """Evaluate model performance"""        test_prophet = self.prepare_data(test_df)        predictions = self.model.predict(test_prophet[['ds']])                mae = mean_absolute_error(test_prophet['y'], predictions['yhat'])        rmse = np.sqrt(mean_squared_error(test_prophet['y'], predictions['yhat']))        mape = np.mean(np.abs((test_prophet['y'] - predictions['yhat']) / test_prophet['y'])) * 100                return {            'MAE': mae,            'RMSE': rmse,            'MAPE': mape        }        def load_model(self):        """Load model from S3"""        model_path = '/tmp/prophet_model.pkl'        self.s3_client.download_file(            self.model_bucket,             'models/forecasting/prophet_model.pkl',            model_path        )        self.model = joblib.load(model_path)        return self.model

    clean_review = review.strip().lower()            # Remove excessive punctuation            clean_review = ' '.join(clean_review.split())            processed.append(clean_review)        return results.append({                    'review': review,                    'sentiment': sentiment['label'],                    'confidence': sentiment['score']                })            except Exception as e:                results.append({                    'review': review,                    'sentiment': 'NEUTRAL',                    'confidence': 0.5,                    'error': str(e)                })                return results        def extract_insights(self, sentiment_results):        """Extract actionable insights from sentiment analysis"""        df = insights['sentiment_distribution'].get('NEGATIVE', 0) / insights['total_reviews']                if negative_ratio > 0.3:            suggestions.append("High negative sentiment detected. Review product quality and customer service.")                if negative_ratio > 0.5:            suggestions.append("Critical: Immediate action required. Consider product recall or major improvements.")                if insights['average_confidence'] < 0.7:            suggestions.append("Mixed signals in reviews. Consider detailed analysis of specific complaint categories.")                return suggestions

# models/pricing/pricing_optimizer.pyimport numpy as npimport pandas as pdfrom sklearn.ensemble import RandomForestRegressorfrom sklearn.preprocessing import StandardScalerimport boto3import joblibclass PricingOptimizer:    def __init__(self):        self.demand_model = RandomForestRegressor(n_estimators=100, random_state=42)        self.scaler = StandardScaler()        self.s3_client = boto3.client('s3')        self.model_bucket = 'sellersense-model-artifacts-bucket'        def prepare_features(self, df):        """Prepare features for pricing model"""        features = df.copy()                # Price-related features        features['price_log'] = np.log(features['price'])        features['competitor_price_ratio'] = features['price'] / features['competitor_avg_price']                # Demand features        features['price_elasticity'] = features['sales'] / features['price']        features['revenue'] = features['sales'] * features['price']                # Temporal features        features['month'] = pd.to_datetime(features['date']).dt.month        features['day_of_week'] = pd.to_datetime(features['date']).dt.dayofweek        features['is_weekend'] = features['day_of_week'].isin([5, 6]).astype(int)                # Seasonal features        features['is_festival_season'] = features['month'].isin([9, 10, 11]).astype(int)        features['is_summer'] = features['month'].isin([3, 4, 5]).astype(int)                return features        def train_demand_model(self, df):        """Train model to predict demand based on price"""        features_df = self.prepare_features(df)                # Feature selection        feature_cols = ['price', 'competitor_avg_price', 'price_log', 'competitor_price_ratio',                       'month', 'day_of_week', 'is_weekend', 'is_festival_season', 'is_summer']                X = features_df[feature_cols]        y = features_df['sales']                # Scale features        X_scaled = self.scaler.fit_transform(X)                # Train model        self.demand_model.fit(X_scaled, y)                # Save models        self._save_models()                return self.demand_model        def predict_demand_at_price(self, price, context_data):        """Predict demand at given price"""        # Create feature vector        features = np.array([[            price,            context_data.get('competitor_avg_price', price * 1.1),            np.log(price),            price / context_data.get('competitor_avg_price', price * 1.1),            context_data.get('month', 6),            context_data.get('day_of_week', 2),            context_data.get('is_weekend', 0),            context_data.get('is_festival_season', 0),            context_data.get('is_summer', 0)        ]])                features_scaled = self.scaler.transform(features)        demand = self.demand_model.predict(features_scaled)[0]                return max(0, demand)  # Ensure non-negative demand        def optimize_price(self, current_price, cost_per_unit, context_data, price_range_factor=0.3):        """Find optimal price to maximize profit"""        # Define price range        min_price = max(cost_per_unit * 1.1, current_price * (1 - price_range_factor))        max_price = current_price * (1 + price_range_factor)                # Test different prices        test_prices = np.linspace(min_price, max_price, 50)        results = []                for price in test_prices:            demand = self.predict_demand_at_price(price, context_data)            revenue = price * demand            profit = (price - cost_per_unit) * demand                        results.append({                'price': price,                'predicted_demand': demand,                'predicted_revenue': revenue,                'predicted_profit': profit,                'margin_percent': ((price - cost_per_unit) / price) * 100            })                results_df = pd.DataFrame(results)                # Find optimal prices        max_profit_idx = results_df['predicted_profit'].idxmax()        max_revenue_idx = results_df['predicted_revenue'].idxmax()                return {            'current_price': current_price,            'optimal_price_profit': results_df.loc[max_profit_idx, 'price'],            'optimal_price_revenue': results_df.loc[max_revenue_idx, 'price'],            'profit_improvement': results_df.loc[max_profit_idx, 'predicted_profit'] -                                 (current_price - cost_per_unit) * self.predict_demand_at_price(current_price, context_data),            'revenue_improvement': results_df.loc[max_revenue_idx, 'predicted_revenue'] -                                  current_price * self.predict_demand_at_price(current_price, context_data),            'price_scenarios': results_df.to_dict('records')        }        def _save_models(self):        """Save trained models to S3"""        # Save demand model        demand_model_path = '/tmp/demand_model.pkl'        joblib.dump(self.demand_model, demand_model_path)                # Save scaler        scaler_path = '/tmp/pricing_scaler.pkl'        joblib.dump(self.scaler, scaler_path)                # Upload to S3        self.s3_client.upload_file(demand_model_path, self.model_bucket, 'models/pricing/demand_model.pkl')        self.s3_client.upload_file(scaler_path, self.model_bucket, 'models/pricing/pricing_scaler.pkl')
